name: Build KoaZa WPF app

on:
  workflow_dispatch:
  push:
    branches: [ "main", "master" ]

jobs:
  build:
    runs-on: windows-latest
    steps:
      - name: Checkout source
        uses: actions/checkout@v4

      - name: Setup MSBuild
        uses: microsoft/setup-msbuild@v2

      - name: Setup NuGet
        uses: NuGet/setup-nuget@v2

      - name: Restore NuGet packages
        run: nuget restore src/KoaZa/KoaZa.sln

      - name: Build Release
        run: msbuild src/KoaZa/KoaZa.sln /m /p:Configuration=Release /p:Platform="Any CPU" /verbosity:minimal

      - name: Prepare portable output folder
        shell: pwsh
        run: |
          $out = "${{ github.workspace }}\artifact\KoaZa"
          New-Item -ItemType Directory -Force -Path $out | Out-Null
          Copy-Item "src\KoaZa\bin\Release\*" $out -Recurse -Force
          Copy-Item "README_FA.md" $out -Force
          if (-not (Test-Path "$out\KoaZa.exe")) { throw "KoaZa.exe was not produced by the build." }

      - name: Upload build artifact
        uses: actions/upload-artifact@v4
        with:
          name: KoaZa-Windows-Build
          path: artifact/KoaZa
          if-no-files-found: error

# KoaZa Football Analytics — بسته منبع و ساخت ابری

این بسته یک **MVP آغازین** برای برنامه دسکتاپ ویندوزی است؛ نسخه نهایی سامانه تحلیل فوتبال نیست. رابط WPF، محاسبه احتمالات پایه با توزیع پواسون، ذخیره پیش‌بینی‌ها در SQLite و گردش‌کار ساخت ابری در GitHub Actions در آن قرار دارد.

## نکته مهم درباره سازگاری ویندوز
- پروژه برای **.NET Framework 4.8** تنظیم شده است.
- برای اجرای برنامه باید .NET Framework 4.8 روی ویندوز نصب باشد؛ این بسته یک نصب‌کننده کامل یا برنامه self-contained نیست.
- Windows 7 فقط در صورت نصب Windows 7 SP1 و به‌روزرسانی‌های لازم ممکن است کار کند. سازگاری TLS/API و اجرای واقعی روی Windows 7 در این بسته آزمایش نشده است.
- کامپایل و اجرای این بسته در این محیط تأیید نشده است. گردش‌کار GitHub Actions برای ساخت روی Windows runner آماده شده است.

## ساخت ابری بدون نصب Visual Studio روی کامپیوتر شما
1. وارد حساب GitHub شوید یا حساب رایگان بسازید.
2. یک مخزن جدید بسازید (Repository)، مثلاً `KoaZa-CloudBuild`.
3. همه فایل‌های داخل این پوشه را در مخزن بارگذاری کنید؛ فایل‌ها و پوشه `.github/workflows` هم باید بارگذاری شوند. ساده‌ترین راه: در صفحه مخزن از **Add file → Upload files** استفاده کنید و فایل‌ها را از حالت ZIP خارج کرده، پوشه‌ها را حفظ کنید.
4. پس از ثبت فایل‌ها، به تب **Actions** بروید. گردش‌کار **Build KoaZa WPF app** باید اجرا شود. اگر درخواست تأیید Actions دیدید، آن را تأیید کنید.
5. وقتی اجرا با وضعیت سبز تمام شد، همان اجرای workflow را باز کنید و از بخش **Artifacts** فایل `KoaZa-Windows-Build` را دانلود کنید.
6. ZIP خروجی را استخراج کنید و `KoaZa.exe` را اجرا کنید. برای اجرای آن، .NET Framework 4.8 لازم است.

اگر ساخت شکست خورد، وارد همان اجرای Actions شوید و بخش قرمز build را باز کنید؛ متن خطا برای رفع مشکل لازم است. موفقیت ساخت ابری به‌تنهایی به معنی تأیید اجرای برنامه روی Windows 7 نیست.

## فایل‌ها
- `src/KoaZa/` — کد برنامه C# و رابط WPF.
- `src/KoaZa/Data/schema.sql` — طرح اولیه SQLite.
- `.github/workflows/build-windows.yml` — ساخت خودکار روی Windows runner و بارگذاری خروجی.

## امکانات همین نسخه
- ورودی نام دو تیم و نرخ گل مورد انتظار هر تیم؛ این نرخ‌ها را کاربر وارد می‌کند و **داده واقعی یا پیش‌بینی خودکار از API نیستند**.
- محاسبه توزیع گل با پواسون و استخراج احتمال 1X2، Over/Under 2.5، BTTS و نتایج دقیق پرتکرار.
- ذخیره ورودی‌ها و احتمالات در SQLite محلی.
- رابط انگلیسی ساده در نسخه اولیه؛ دوزبانه‌سازی کامل، دریافت خودکار API، داده تاریخی، ارزیابی Walk-forward و مدیریت کلید API هنوز پیاده‌سازی نشده‌اند.

## حریم خصوصی کلید API
این نسخه هنوز کلید API نمی‌خواهد. کلیدها را در کد یا مخزن GitHub قرار ندهید.

using System;
using System.IO;
using System.Data.SQLite;
namespace KoaZa.Data
{
    public sealed class DatabaseService
    {
        private readonly string _connectionString;
        public DatabaseService()
        {
            string folder = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "Data");
            Directory.CreateDirectory(folder);
            string path = Path.Combine(folder, "koaza.db");
            _connectionString = "Data Source=" + path + ";Version=3;";
            if (!File.Exists(path)) SQLiteConnection.CreateFile(path);
            Initialize();
        }
        private void Initialize()
        {
            using (var con = new SQLiteConnection(_connectionString))
            {
                con.Open();
                using (var cmd = new SQLiteCommand("CREATE TABLE IF NOT EXISTS Predictions (Id INTEGER PRIMARY KEY AUTOINCREMENT, CreatedUtc TEXT NOT NULL, HomeTeam TEXT NOT NULL, AwayTeam TEXT NOT NULL, HomeGoalRate REAL NOT NULL, AwayGoalRate REAL NOT NULL, HomeWin REAL NOT NULL, DrawProbability REAL NOT NULL, AwayWin REAL NOT NULL, Over25 REAL NOT NULL, Under25 REAL NOT NULL, BothTeamsScore REAL NOT NULL, ModelVersion TEXT NOT NULL);", con)) cmd.ExecuteNonQuery();
            }
        }
        public void SavePrediction(string home, string away, double homeRate, double awayRate, Services.MatchProbabilities p)
        {
            using (var con = new SQLiteConnection(_connectionString))
            {
                con.Open();
                using (var cmd = new SQLiteCommand("INSERT INTO Predictions (CreatedUtc, HomeTeam, AwayTeam, HomeGoalRate, AwayGoalRate, HomeWin, DrawProbability, AwayWin, Over25, Under25, BothTeamsScore, ModelVersion) VALUES (@date,@home,@away,@hr,@ar,@hw,@draw,@aw,@over,@under,@btts,@version);", con))
                {
                    cmd.Parameters.AddWithValue("@date", DateTime.UtcNow.ToString("o")); cmd.Parameters.AddWithValue("@home", home); cmd.Parameters.AddWithValue("@away", away);
                    cmd.Parameters.AddWithValue("@hr", homeRate); cmd.Parameters.AddWithValue("@ar", awayRate); cmd.Parameters.AddWithValue("@hw", p.HomeWin); cmd.Parameters.AddWithValue("@draw", p.Draw); cmd.Parameters.AddWithValue("@aw", p.AwayWin);
                    cmd.Parameters.AddWithValue("@over", p.Over25); cmd.Parameters.AddWithValue("@under", p.Under25); cmd.Parameters.AddWithValue("@btts", p.BothTeamsScore); cmd.Parameters.AddWithValue("@version", "poisson-baseline-0.1"); cmd.ExecuteNonQuery();
                }
            }
        }
    }
}

CREATE TABLE IF NOT EXISTS Predictions (
  Id INTEGER PRIMARY KEY AUTOINCREMENT,
  CreatedUtc TEXT NOT NULL,
  HomeTeam TEXT NOT NULL,
  AwayTeam TEXT NOT NULL,
  HomeGoalRate REAL NOT NULL CHECK(HomeGoalRate >= 0),
  AwayGoalRate REAL NOT NULL CHECK(AwayGoalRate >= 0),
  HomeWin REAL NOT NULL CHECK(HomeWin BETWEEN 0 AND 1),
  DrawProbability REAL NOT NULL CHECK(DrawProbability BETWEEN 0 AND 1),
  AwayWin REAL NOT NULL CHECK(AwayWin BETWEEN 0 AND 1),
  Over25 REAL NOT NULL CHECK(Over25 BETWEEN 0 AND 1),
  Under25 REAL NOT NULL CHECK(Under25 BETWEEN 0 AND 1),
  BothTeamsScore REAL NOT NULL CHECK(BothTeamsScore BETWEEN 0 AND 1),
  ModelVersion TEXT NOT NULL
);
CREATE INDEX IF NOT EXISTS IX_Predictions_CreatedUtc ON Predictions(CreatedUtc);

using System;
using System.Globalization;
using System.Windows;
using KoaZa.Data;
using KoaZa.Services;
namespace KoaZa
{
    public partial class MainWindow : Window
    {
        private readonly DatabaseService _database;
        public MainWindow()
        {
            InitializeComponent();
            try { _database = new DatabaseService(); }
            catch (Exception ex) { MessageBox.Show("Could not initialize the local database.\n" + ex.Message, "Database error", MessageBoxButton.OK, MessageBoxImage.Warning); }
        }
        private void AnalyzeButton_Click(object sender, RoutedEventArgs e)
        {
            double homeRate, awayRate;
            if (string.IsNullOrWhiteSpace(HomeTeamBox.Text) || string.IsNullOrWhiteSpace(AwayTeamBox.Text) ||
                !double.TryParse(HomeRateBox.Text, NumberStyles.Float, CultureInfo.InvariantCulture, out homeRate) ||
                !double.TryParse(AwayRateBox.Text, NumberStyles.Float, CultureInfo.InvariantCulture, out awayRate) || homeRate < 0 || awayRate < 0 || homeRate > 8 || awayRate > 8)
            { MessageBox.Show("Enter both team names and valid expected-goal rates from 0 to 8 (use a dot for decimals).", "Check inputs", MessageBoxButton.OK, MessageBoxImage.Information); return; }
            try
            {
                var p = PoissonModel.Analyze(homeRate, awayRate);
                MatchTitle.Text = HomeTeamBox.Text.Trim() + " vs " + AwayTeamBox.Text.Trim();
                HomeWinText.Text = "Home win: " + Percent(p.HomeWin); DrawText.Text = "Draw: " + Percent(p.Draw); AwayWinText.Text = "Away win: " + Percent(p.AwayWin);
                OverText.Text = "Over 2.5 goals: " + Percent(p.Over25); UnderText.Text = "Under 2.5 goals: " + Percent(p.Under25); BttsText.Text = "Both teams to score: " + Percent(p.BothTeamsScore);
                ScoresGrid.ItemsSource = p.TopScores;
                if (_database != null) { _database.SavePrediction(HomeTeamBox.Text.Trim(), AwayTeamBox.Text.Trim(), homeRate, awayRate, p); QualityText.Text = "Saved to local SQLite • Rates are user-entered, not verified data."; }
                else QualityText.Text = "Not saved: database initialization failed. Rates are user-entered only.";
            }
            catch (Exception ex) { MessageBox.Show("Analysis failed:\n" + ex.Message, "Error", MessageBoxButton.OK, MessageBoxImage.Error); }
        }
        private static string Percent(double value) { return (value * 100).ToString("0.0", CultureInfo.InvariantCulture) + "%"; }
    }
}

using System;
using System.Collections.Generic;
using System.Linq;
namespace KoaZa.Services
{
    public sealed class ScoreProbability
    {
        public string Score { get; set; }
        public string Probability { get; set; }
        public double Value { get; set; }
    }
    public sealed class MatchProbabilities
    {
        public double HomeWin { get; set; }
        public double Draw { get; set; }
        public double AwayWin { get; set; }
        public double Over25 { get; set; }
        public double Under25 { get; set; }
        public double BothTeamsScore { get; set; }
        public List<ScoreProbability> TopScores { get; set; }
    }
    public static class PoissonModel
    {
        private static double Factorial(int n) { double v = 1.0; for (int i = 2; i <= n; i++) v *= i; return v; }
        public static double PoissonPmf(int goals, double lambda)
        {
            if (lambda < 0 || double.IsNaN(lambda) || double.IsInfinity(lambda)) throw new ArgumentOutOfRangeException("lambda");
            return Math.Exp(-lambda) * Math.Pow(lambda, goals) / Factorial(goals);
        }
        public static MatchProbabilities Analyze(double homeRate, double awayRate)
        {
            if (homeRate < 0 || awayRate < 0 || homeRate > 8 || awayRate > 8) throw new ArgumentOutOfRangeException("Rates must be between 0 and 8.");
            const int maxGoals = 12;
            var hp = Enumerable.Range(0, maxGoals + 1).Select(i => PoissonPmf(i, homeRate)).ToArray();
            var ap = Enumerable.Range(0, maxGoals + 1).Select(i => PoissonPmf(i, awayRate)).ToArray();
            double hw = 0, draw = 0, aw = 0, over = 0, btts = 0;
            var scores = new List<ScoreProbability>();
            for (int h = 0; h <= maxGoals; h++)
            for (int a = 0; a <= maxGoals; a++)
            {
                double p = hp[h] * ap[a];
                if (h > a) hw += p; else if (h == a) draw += p; else aw += p;
                if (h + a >= 3) over += p;
                if (h > 0 && a > 0) btts += p;
                scores.Add(new ScoreProbability { Score = h + " - " + a, Value = p, Probability = (p * 100).ToString("0.00") + "%" });
            }
            // Normalize 1X2 for the negligible probability mass above the 12-goal truncation.
            double sum = hw + draw + aw;
            return new MatchProbabilities { HomeWin = hw / sum, Draw = draw / sum, AwayWin = aw / sum,
                Over25 = over, Under25 = 1 - over, BothTeamsScore = btts,
                TopScores = scores.OrderByDescending(s => s.Value).Take(10).ToList() };
        }
    }
}
